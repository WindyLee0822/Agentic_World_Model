import re
import requests
import pandas as pd
from bs4 import BeautifulSoup

URL = "http://server:8000"


def extract_first_number(text, cast=float):
    m = re.search(r"[-+]?\d*\.?\d+", text or "")
    if not m:
        return None
    return cast(float(m.group())) if cast is int else cast(m.group())


def main():
    response = requests.get(URL, timeout=10)
    response.raise_for_status()
    soup = BeautifulSoup(response.text, "html.parser")

    books = []
    for book in soup.select("div.book"):
        title = book.select_one("div.book-title")
        author = book.select_one("div.book-author")
        year = book.select_one("span.book-year")
        rating = book.select_one("span.book-rating")
        price = book.select_one("span.book-price")

        title_text = title.get_text(strip=True) if title else ""
        author_text = author.get_text(strip=True) if author else ""
        author_text = re.sub(r"^by\s+", "", author_text, flags=re.I)
        year_val = extract_first_number(year.get_text(" ", strip=True) if year else "", int)
        rating_val = extract_first_number(rating.get_text(" ", strip=True) if rating else "", float)
        price_val = extract_first_number(price.get_text(" ", strip=True) if price else "", float)

        books.append({
            "title": title_text,
            "author": author_text,
            "year": year_val,
            "rating": rating_val,
            "price": price_val,
        })

    df = pd.DataFrame(books, columns=["title", "author", "year", "rating", "price"])
    df.to_csv("/app/books.csv", index=False)

    total_books = len(df)
    avg_price = df["price"].dropna().mean() if total_books else 0
    highest = df.sort_values(["rating", "title"], ascending=[False, True]).iloc[0] if total_books else None

    with open("/app/report.txt", "w", encoding="utf-8") as f:
        f.write(f"Total number of books scraped: {total_books}\n")
